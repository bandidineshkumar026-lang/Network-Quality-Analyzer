  <section class="doc" aria-labelledby="h-code">
    <h2 id="h-code">Code description</h2>
    <p>The analyser is one HTML file with plain JavaScript and no libraries. The tests request two public Cloudflare speed-test URLs, so no support files or server of your own are needed.</p>
    <dl class="files">
      <dt>index.html</dt><dd>Interface, measurement code, chart and scoring. This is the only file.</dd>
      <dt>PING_URL</dt><dd>Returns an empty response. Requesting it repeatedly gives the round-trip time.</dd>
      <dt>PAYLOAD_URL</dt><dd>Returns 5 MB of generated bytes for the download test.</dd>
    </dl>

    <h3>How a test runs</h3>
    <ol>
      <li>One warm-up request opens the TCP and TLS connection, so connection setup does not inflate the first sample.</li>
      <li>30 timed requests to <code>PING_URL</code> are sent 120 ms apart. Each has a 2 second timeout.</li>
      <li>Latency, jitter and packet loss are calculated from those 30 results.</li>
      <li><code>PAYLOAD_URL</code> is downloaded up to three times. A run is cut off after 10 seconds, and a slow link stops after one run.</li>
      <li>The median download speed and the three values above are combined into a score from 0 to 100.</li>
    </ol>

    <h3>Latency (round-trip time)</h3>
    <p>Latency is the time from sending a request to receiving the full response. It adds up propagation, transmission, queuing and processing delay. The page reports the median of the successful probes, so one slow probe does not move the result. Timing uses <code>performance.now()</code>, which has sub-millisecond resolution. A probe that returns null was lost.</p>
    <pre><code id="code-probe"></code></pre>

    <h3>Jitter</h3>
    <p>Jitter is the variation in delay, mostly caused by queuing at routers. The page uses the mean absolute difference between consecutive round-trip times: <code>jitter = sum(|RTT[i] - RTT[i-1]|) / (n - 1)</code>. RFC 3550 uses a smoothed version of the same idea for RTP voice and video streams.</p>
    <pre><code id="code-jitter"></code></pre>

    <h3>Packet loss</h3>
    <p>A browser cannot send raw packets, so loss is counted per request. A probe is lost when it times out after 2 seconds or returns an HTTP error. The formula is <code>loss % = lost probes / probes sent * 100</code>.</p>

    <h3>Download speed</h3>
    <p>Speed is the number of bits received divided by the time taken: <code>Mbps = bytes * 8 / seconds / 1,000,000</code>. The clock starts when the first chunk arrives, so latency is not counted as transfer time. Progress is read chunk by chunk from the response stream, which drives the live number during the test.</p>
    <pre><code id="code-download"></code></pre>

    <h3>Quality score</h3>
    <p>Each measurement is mapped to a value between 0 and 1 using a best and a worst limit, then weighted: latency 30%, jitter 20%, packet loss 25%, download speed 25%. Speed uses a log scale because going from 1 to 10 Mbps matters more than going from 91 to 100. The chips beside the score compare your numbers with the limits printed under each use. A use is Suitable when every limit is met, Marginal when the worst value is within 50% of its limit, and Not suitable beyond that.</p>
    <pre><code id="code-score"></code></pre>

    <h3>Limits of the method</h3>
    <ul>
      <li>It measures the path to Cloudflare's test server over HTTPS (TCP, TLS and HTTP), not an ICMP ping. Browsers cannot send ICMP or raw packets.</li>
      <li>TCP retransmits lost packets, so request-level loss reads lower than loss measured with UDP or ICMP.</li>
      <li>Speed comes from a single connection, so it can be lower than a multi-connection speed test.</li>
      <li>Results change with Wi-Fi signal, other devices on the network and VPNs. Run the test a few times.</li>
    </ul>

    <h3>Extending the project</h3>
    <p>A small server in Python or Node would add what a browser cannot do: an upload test, a UDP loss test and traceroute. Saving each result with a timestamp would give a history chart.</p>

const PROBES = 30, GAP_MS = 120, TIMEOUT_MS = 2000, DL_RUNS = 3, MAX_DL_MS = 10000;
// Public Cloudflare speed-test endpoints (they allow cross-origin requests).
// To test against your own server instead, point these at a tiny file and a ~5 MB file you host.
const PING_URL = 'https://speed.cloudflare.com/__down?bytes=0';
const PAYLOAD_URL = 'https://speed.cloudflare.com/__down?bytes=5000000';

async function probeOnce(url, timeoutMs) {
  // One timed request. Returns the round-trip time in ms, or null if the probe was lost.
  const ctl = new AbortController();
  const timer = setTimeout(() => ctl.abort(), timeoutMs);
  const t0 = performance.now();
  try {
    const res = await fetch(url, { cache: 'no-store', signal: ctl.signal });
    if (!res.ok) throw new Error('HTTP ' + res.status);
    await res.arrayBuffer();
    return performance.now() - t0;
  } catch (e) {
    return null; // timeout or HTTP error counts as a lost probe
  } finally {
    clearTimeout(timer);
  }
}

function median(values) {
  const s = [...values].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m] : (s[m - 1] + s[m]) / 2;
}

function jitter(rtts) {
  // Mean absolute difference between consecutive round-trip times.
  if (rtts.length < 2) return 0;
  let sum = 0;
  for (let i = 1; i < rtts.length; i++) sum += Math.abs(rtts[i] - rtts[i - 1]);
  return sum / (rtts.length - 1);
}

async function downloadOnce(url, onTick) {
  // Streams the payload and returns throughput in Mbit/s.
  const ctl = new AbortController();
  const res = await fetch(url, { cache: 'no-store', signal: ctl.signal });
  if (!res.ok) throw new Error('HTTP ' + res.status);
  const reader = res.body.getReader();
  let bytes = 0, tFirst = 0, tLast = 0;
  try {
    for (;;) {
      const { done, value } = await reader.read();
      const now = performance.now();
      if (done) break;
      if (!tFirst) { tFirst = now; continue; } // start the clock at the first byte
      bytes += value.length;
      tLast = now;
      onTick(bytes * 8 / ((now - tFirst) / 1000) / 1e6);
      if (now - tFirst > MAX_DL_MS) { ctl.abort(); break; } // slow link: stop early
    }
  } catch (e) { /* aborted */ }
  if (!tLast || tLast === tFirst) throw new Error('No data received');
  return bytes * 8 / ((tLast - tFirst) / 1000) / 1e6;
}

const clamp01 = x => Math.max(0, Math.min(1, x));

function score(m) {
  const sLat = 1 - clamp01((m.lat - 20) / 280);              // 1 at 20 ms or less, 0 at 300 ms
  const sJit = 1 - clamp01((m.jit - 5) / 45);                // 1 at 5 ms or less, 0 at 50 ms
  const sLoss = 1 - clamp01(m.loss / 5);                     // 1 at 0 %, 0 at 5 % or more
  const sSpd = clamp01(Math.log10(Math.max(m.mbps, 1)) / 2); // 0 at 1 Mbps, 1 at 100 Mbps
  return Math.round(100 * (0.30 * sLat + 0.20 * sJit + 0.25 * sLoss + 0.25 * sSpd));
}

function fit(u, m, mbps) {
  const worst = Math.max(m.lat / u.lat, m.jit / u.jit, m.loss / u.loss, u.mbps / mbps);
  return worst <= 1 ? 'ok' : worst <= 1.5 ? 'mid' : 'bad';
}

function summarize(samples) {
  const ok = samples.filter(v => v !== null);
  const sent = samples.length, lost = sent - ok.length;
  return {
    sent, lost, loss: sent ? lost / sent * 100 : 0,
    lat: ok.length ? median(ok) : null,
    min: ok.length ? Math.min(...ok) : null,
    max: ok.length ? Math.max(...ok) : null,
    jit: jitter(ok)
  };
}

function niceCeil(x) {
  const steps = [20, 50, 100, 200, 500, 1000, 2000, 5000];
  return steps.find(s => s >= x) || Math.ceil(x / 1000) * 1000;
}

function renderTrace(s) {
  lastSamples = s;
  const W = Math.max(280, Math.floor($('trace-box').clientWidth || 640));
  const H = W < 480 ? 190 : 230;
  const L = 58, R = 12, T = 14, B = 26, pw = W - L - R, ph = H - T - B;
  const ok = s.filter(v => v !== null);
  const top = niceCeil(Math.max(20, ...ok) * 1.1);
  const x = i => L + (i / (PROBES - 1)) * pw;
  const y = v => T + ph - (v / top) * ph;
  let out = '';
  [0, top / 2, top].forEach((t, k) => {
    out += '<line class="grid" x1="' + L + '" x2="' + (W - R) + '" y1="' + y(t).toFixed(1) + '" y2="' + y(t).toFixed(1) + '"/>' +
      '<text class="tick" x="' + (L - 8) + '" y="' + (y(t) + 4).toFixed(1) + '" text-anchor="end">' + t + (k === 2 ? ' ms' : '') + '</text>';
  });
  if (ok.length > 1) {
    const my = y(median(ok)).toFixed(1);
    out += '<line class="med" x1="' + L + '" x2="' + (W - R) + '" y1="' + my + '" y2="' + my + '"/>';
  }
  let d = '', pen = false;
  s.forEach((v, i) => {
    if (v === null) { pen = false; return; }
    d += (pen ? 'L' : 'M') + x(i).toFixed(1) + ' ' + y(v).toFixed(1);
    pen = true;
  });
  out += '<path class="line" d="' + d + '"/>';
  s.forEach((v, i) => {
    if (v === null) {
      const cx = x(i), cy = T + 8;
      out += '<line class="lostline" x1="' + cx.toFixed(1) + '" x2="' + cx.toFixed(1) + '" y1="' + T + '" y2="' + (T + ph) + '"/>' +
        '<path class="lost" d="M' + (cx - 4).toFixed(1) + ' ' + (cy - 4) + 'l8 8M' + (cx - 4).toFixed(1) + ' ' + (cy + 4) + 'l8 -8"/>';
    } else {
      out += '<circle class="dot" cx="' + x(i).toFixed(1) + '" cy="' + y(v).toFixed(1) + '" r="2.6"/>';
    }
  });
  out += '<text class="tick" x="' + L + '" y="' + (H - 6) + '">probe 1</text>' +
    '<text class="tick" x="' + (W - R) + '" y="' + (H - 6) + '" text-anchor="end">' + PROBES + '</text>';
  const svg = $('trace');
  svg.setAttribute('viewBox', '0 0 ' + W + ' ' + H);
  svg.setAttribute('height', H);
  svg.setAttribute('aria-label', ok.length
    ? 'Round-trip time per probe, median ' + median(ok).toFixed(0) + ' milliseconds, ' + (s.length - ok.length) + ' lost'
    : 'Round-trip time per probe, no data yet');
  svg.innerHTML = out;
}

function clearVerdict() {
  $('score').textContent = '—';
  const g = $('grade'); g.textContent = 'Measuring'; g.className = 'grade';
  $('uses').innerHTML = USES.map(u =>
    '<li><div><b>' + u.name + '</b><small>' + needText(u) + '</small></div><span class="chip wait">Pending</span></li>').join('');
}

function needText(u) {
  return '≥ ' + u.mbps + ' Mbps · ≤ ' + u.lat + ' ms · jitter ≤ ' + u.jit + ' ms · loss ≤ ' + u.loss + '%';
}

function showVerdict(m, mbps) {
  const sc = score({ lat: m.lat, jit: m.jit, loss: m.loss, mbps });
  const grade = sc >= 85 ? ['Excellent', 'g-ex'] : sc >= 70 ? ['Good', 'g-good'] : sc >= 50 ? ['Fair', 'g-fair'] : ['Poor', 'g-poor'];
  $('score').textContent = sc;
  const g = $('grade'); g.textContent = grade[0]; g.className = 'grade ' + grade[1];
  const label = { ok: 'Suitable', mid: 'Marginal', bad: 'Not suitable' };
  $('uses').innerHTML = USES.map(u => {
    const f = fit(u, m, mbps);
    return '<li><div><b>' + u.name + '</b><small>' + needText(u) + '</small></div><span class="chip ' + f + '">' + label[f] + '</span></li>';
  }).join('');
  return grade[0];
}

function setRunning(on) {
  const b = $('run');
  b.textContent = on ? 'Stop' : 'Run again';
  b.classList.toggle('is-running', on);
}

async function runTest() {
  if (running) { cancel = true; setStatus('Stopping…'); return; }
  running = true; cancel = false;
  const samples = [];
  setRunning(true);
  $('sample-tag').hidden = true;
  showReadouts(null); setDl(null, 'median of runs'); clearVerdict(); renderTrace(samples); setProgress(0);
  const total = PROBES + DL_RUNS;
  try {
    setStatus('Opening connection…');
    await probeOnce(PING_URL, TIMEOUT_MS); // warm-up, not recorded
    for (let i = 0; i < PROBES && !cancel; i++) {
      setStatus('Latency: probe ' + (i + 1) + ' of ' + PROBES);
      samples.push(await probeOnce(PING_URL, TIMEOUT_MS));
      renderTrace(samples); showReadouts(summarize(samples)); setProgress((i + 1) / total);
      await sleep(GAP_MS);
    }
    const m = summarize(samples);
    if (cancel) { setStatus('Stopped. Partial results are shown.'); return; }
    if (m.lat === null) { setStatus('No reply from the test server. Check your connection and try again.'); return; }

    const runs = [];
    for (let r = 0; r < DL_RUNS && !cancel; r++) {
      setStatus('Download: run ' + (r + 1) + ' of ' + DL_RUNS);
      try {
        const mbps = await downloadOnce(PAYLOAD_URL, v => setDl(v, 'run ' + (r + 1) + ' of ' + DL_RUNS + ', live'));
        runs.push(mbps);
        setDl(median(runs), 'median of ' + runs.length + (runs.length === 1 ? ' run' : ' runs'));
        setProgress((PROBES + r + 1) / total);
        if (mbps < 5) break; // slow link: one run is enough
      } catch (e) { break; }
    }
    if (cancel) { setStatus('Stopped. Partial results are shown.'); return; }
    if (!runs.length) { setDl(null, 'download failed'); setStatus('The download test failed. Latency results are shown.'); return; }
    const verdict = showVerdict(m, median(runs));
    setStatus('Done. This connection rates ' + verdict.toLowerCase() + '.');
  } catch (e) {
    setStatus('The test could not finish. Try again.');
  } finally {
    running = false; setRunning(false); setProgress(0);
  }
}

// ---- sample reading shown before the first run --------------------------------
function sampleRtts() {
  let seed = 7;
  const rnd = () => (seed = (seed * 16807) % 2147483647) / 2147483647;
  return Array.from({ length: PROBES }, (_, i) =>
    i === 17 ? null : Math.round((36 + rnd() * 10 + (i % 9 === 4 ? 14 : 0)) * 10) / 10);
}

function init() {
  [['code-probe', probeOnce], ['code-jitter', jitter], ['code-download', downloadOnce], ['code-score', score]]
    .forEach(p => { $(p[0]).textContent = p[1].toString(); });

  const s = sampleRtts(), m = summarize(s);
  renderTrace(s); showReadouts(m); setDl(87.4, 'sample, median of 3 runs'); showVerdict(m, 87.4);

  const c = navigator.connection;
  if (c && (c.effectiveType || c.downlink || c.rtt)) {
    const parts = [];
    if (c.effectiveType) parts.push(c.effectiveType + ' link');
    if (c.downlink) parts.push(c.downlink + ' Mbps downlink');
    if (c.rtt) parts.push(c.rtt + ' ms RTT');
    $('link-info').textContent = 'Your browser reports: ' + parts.join(' · ') + '.';
    $('link-info').hidden = false;
  }
  $('run').addEventListener('click', runTest);
  if (window.ResizeObserver) new ResizeObserver(() => renderTrace(lastSamples)).observe($('trace-box'));
}
init();
